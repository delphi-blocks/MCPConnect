unit %UNIT_NAME%;

interface

uses
  System.SysUtils, System.Classes,

  MCPConnect.MCP.Attributes;

type
  /// <summary>
  ///   Every tool of this class is exposed as "sample_&lt;toolname&gt;"
  /// </summary>
  [McpScope('sample')]
  TSampleTool = class
  public
    [McpTool('echo', 'Echoes back the message received from the client')]
    function Echo(
      [McpParam('message', 'The message to echo back')] const AMessage: string): string;

    [McpTool('sum', 'Adds two numbers and returns the result')]
    function Sum(
      [McpParam('first', 'The first addend')] AFirst: Integer;
      [McpParam('second', 'The second addend')] ASecond: Integer): Integer;
  end;

  TSampleResource = class
  public
    [McpResource('server-info', 'text://info', 'text/plain', 'General information about this server')]
    function GetInfo: string;

    [McpTemplate('greeting', 'text://greeting/{name}', 'text/plain', 'A greeting for the given name')]
    function GetGreeting(
      [McpParam('name', 'The name to greet')] const AName: string): string;
  end;

  TSamplePrompts = class
  public
    [MCPPrompt('summarize', 'Summarize', 'Asks the model to summarize a text')]
    function Summarize(
      [MCPArgument('text', 'The text to summarize', 'required')] const AText: string): string;
  end;

implementation

{ TSampleTool }

function TSampleTool.Echo(const AMessage: string): string;
begin
  Result := AMessage;
end;

function TSampleTool.Sum(AFirst, ASecond: Integer): Integer;
begin
  Result := AFirst + ASecond;
end;

{ TSampleResource }

function TSampleResource.GetInfo: string;
begin
  Result := 'This MCP server has been generated by the MCPConnect project wizard.';
end;

function TSampleResource.GetGreeting(const AName: string): string;
begin
  Result := Format('Hello, %s!', [AName]);
end;

{ TSamplePrompts }

function TSamplePrompts.Summarize(const AText: string): string;
begin
  Result := Format('Summarize the following text in a few sentences:'#13#10#13#10'%s', [AText]);
end;

end.
